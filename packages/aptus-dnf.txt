# Packages for aptus as the main shell (Fedora, dnf). One name per line; "## group" starts a group; text after # is a note.
# bin/aptus-packages reads this file. Nothing here is installed by reading it.

## required
hyprland              # compositor (Lua config)
quickshell            # the shell runtime, 0.3.2 or newer (Fedora or COPR)
hyprsunset            # night light
brightnessctl         # screen brightness
playerctl             # music state and keys
grim                  # screenshots
slurp                 # screenshot regions
wl-clipboard          # wl-copy, wl-paste
cliphist              # clipboard history
wf-recorder           # screen recording
libnotify             # notify-send
curl                  # weather and web search
python3               # AI usage helper, patch scripts
ImageMagick           # wallpaper colours (magick)
power-profiles-daemon # power profiles
NetworkManager        # Wi-Fi (nmcli)
iio-sensor-proxy      # rotation and light sensor
bluez                 # Bluetooth
pipewire              # audio
wireplumber           # audio sessions
xdg-utils             # xdg-open

## session
hypridle              # idle and lock timing
hyprlock              # backup lock screen
polkit                # the shell shows the password prompt

## backup
noctalia-nightly      # the backup shell, started by shell-switch if aptus does not come up

## optional
ghostty               # terminal
nemo                  # file manager
jetbrains-mono-nerd-fonts # monospace font the shell prefers
cage                  # only for the aptus login screen (--greeter)
greetd                # only for the aptus login screen (--greeter)
